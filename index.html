async def event_loop(self):
    audio_frame = await self.mic.capture()
    if self.mic.detect_speech(audio_frame):
        query = await self.transcribe(audio_frame)
        intent = self.route_intent(query)
        result = await self.os_bridge.dispatch(intent)
